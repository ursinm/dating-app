## 📝 Пример комментариев

### До:
```typescript
import { useState } from 'react';

function Component() {
  const [data, setData] = useState([]);
  
  useEffect(() => {
    loadData();
  }, []);
  
  return <div>{data}</div>;
}
```

### После:
```typescript
// Импорт хука состояния из React
import { useState } from 'react';

/**
 * Компонент для отображения данных
 * Загружает и показывает список элементов
 */
function Component() {
  // Состояние для хранения загруженных данных
  const [data, setData] = useState([]);
  
  // Эффект для загрузки данных при монтировании компонента
  useEffect(() => {
    loadData(); // Вызываем функцию загрузки
  }, []);
  
  // Рендерим список данных
  return <div>{data}</div>;
}
```
