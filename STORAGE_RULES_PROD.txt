// ПРАВИЛА ДЛЯ ПРОДАКШЕНА (безопасные)
// Используйте после тестирования!

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Сообщения (фото и голосовые)
    match /messages/{chatId}/{fileName} {
      // Чтение: только для аутентифицированных
      allow read: if request.auth != null;
      
      // Запись: только для аутентифицированных, макс 10MB
      allow write: if request.auth != null 
                   && request.resource.size < 10 * 1024 * 1024;
    }
    
    // Аватары пользователей
    match /users/{userId}/{fileName} {
      allow read: if true;
      allow write: if request.auth != null 
                   && request.auth.uid == userId
                   && request.resource.size < 5 * 1024 * 1024;
    }
  }
}
